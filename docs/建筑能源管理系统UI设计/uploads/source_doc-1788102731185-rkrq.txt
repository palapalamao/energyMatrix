//
// 点位规范
// 三层点位约定（能耗数据链的关键）：
//   L1 raw    累积读数（表底数）  total + sensor        —— 直采，禁止修改
//   L2 delta  区间增量           emDelta + his rollup  —— 由 raw 差分，含翻转/换表补偿
//   L3 index  归一化指标         emNormalized          —— 业务层生成（见业务对象模型）
// 所有 L1/L2 点位强制 his + hisInterpolate:"linear" + tz 继承 site。
//

emDelta: Marker <doc:"区间增量点，由累积量差分得到">

// --- 电 -------------------------------------------------------------------

  unit: "kWh"
  his
}

  emTouPeriod: Str <doc:"sharp | peak | flat | valley">
  unit: "kWh"
  his
}
emTouPeriod: Str <doc:"分时电价时段">

  unit: "kW"

}

}
